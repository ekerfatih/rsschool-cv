[Multiples of 3 or 5](https://www.codewars.com/kata/514b92a657cdc65150000006/train/javascript)

```javascript
function solution(number){
  if(number < 0) return 0;
  let total = 0;
  for(let i = 0; i < number; i++){
    if(i % 3===0 || i % 5===0){
      total += i;
      continue;
    }
  }
```
